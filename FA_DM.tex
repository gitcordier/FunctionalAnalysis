    \IfNoValueTF{#1}

% #1 = lower bound, #2 = upper bound.
% The sign takes the current style, the bounds the matching script style. \mathchoice typesets all 4 styles: only the
% first one saves the MathML of the bounds.
\newcommand{\symint@do}[2]{%
  \global\advance\symint@n\@ne
  \symint@annotate{#1}{#2}{%
    \mathop{\mathchoice
      {\symint@build\symint@save\displaystyle\scriptstyle{#1}{#2}}%
      {\symint@build\@gobble\textstyle\scriptstyle{#1}{#2}}%
      {\symint@build\@gobble\scriptstyle\scriptscriptstyle{#1}{#2}}%
      {\symint@build\@gobble\scriptscriptstyle\scriptscriptstyle{#1}{#2}}}\nolimits}}

% #1 = \symint@save or \@gobble, #2 = style of the sign, #3 = style of the bounds, #4 = lower bound, #5 = upper bound
\newcommand{\symint@build}[5]{%
  \begingroup
  \setbox\symint@sign\hbox{$\m@th\symint@ignore#2\intReg\nolimits$}%
  \symint@w=\wd\symint@sign
  \symint@c=\dimexpr.5\symint@w+\symintcx\relax
  \setbox\symint@tmp\hbox{$\m@th\symint@ignore#2\vcenter{\hrule height0pt depth0pt}$}%
  \symint@ax=\ht\symint@tmp                       % math axis
  \setbox\symint@tmp\hbox{$\m@th\symint@ignore#2\intReg\nolimits^{\vrule width0pt height 100em depth0pt}$}%
  \symint@u=\dimexpr\ht\symint@tmp-100em\relax    % baseline of a superscript on \intReg
  \setbox\symint@tmp\hbox{$\m@th\symint@ignore#3#5$}\symint@wu=\wd\symint@tmp
  \setbox\symint@tmp\hbox{$\m@th\symint@ignore#3#4$}\symint@wl=\wd\symint@tmp
  % distances from the center of the sign to the centers of the bounds
  \symint@sh=\dimexpr.5\symint@w+.5\symint@wu+\symintgap\relax
  \symint@sl=\dimexpr.5\symint@w+.5\symint@wl+\symintgap\relax
  % same height and depth for both bounds, so that they mirror each other
  \setbox\symint@tmp\hbox{$\m@th\symint@ignore#3\mathstrut\vphantom{#4#5}$}%
  \setbox\symint@hi\hbox to 0pt{\hss\copy\symint@tmp$\m@th\symint@ignore#1{hi}#3#5$\hss}%
  \setbox\symint@lo\hbox to 0pt{\hss\copy\symint@tmp$\m@th\symint@ignore#1{lo}#3#4$\hss}%
  \symint@d=\dp\symint@hi
  \symint@ovr=\dimexpr\symint@c+\symint@sh+.5\symint@wu-\symint@w\relax
  \symint@ovl=\dimexpr\symint@sl-\symint@c+.5\symint@wl\relax
  % no room for a missing bound
  \ifdim\symint@wu=0pt \symint@ovr=0pt \fi
  \ifdim\symint@wl=0pt \symint@ovl=0pt \fi
  \ifdim\symint@ovr<0pt \symint@ovr=0pt \fi
  \ifdim\symint@ovl<0pt \symint@ovl=0pt \fi
  \hbox{%
    \kern\symint@ovl
    \copy\symint@sign
    \hbox to 0pt{%
      \kern-\symint@w
      $\m@th\symint@ignore#2\vcenter{\vbox{\offinterlineskip
        \moveright\dimexpr\symint@c+\symint@sh\relax\box\symint@hi
        \kern\dimexpr2\symint@u-2\symint@ax-2\symint@d\relax
        \moveright\dimexpr\symint@c-\symint@sl\relax\box\symint@lo}}$%
      \hss}%
    \kern\symint@ovr
  }%
  \endgroup}
\makeatother
  
% #1 = domain (optional), #2 = integrand, #3 = measure (optional): \intLebesgue[K]{u \cdot f}[\beta].
% The trailing optional #3 takes any [ that follows the integrand, so \intLebesgue[K]{f} [a] reads [a] as the measure.
% This never occurs by design: an integral inside a computation is wrapped in parentheses or brackets, e.g.
% \bigl[\intLebesgue[K]{f}[\beta]\bigr], so a [ right after the integrand is a typo.

  \ensuremath{%
    \intReg
    \IfValueT{#1}{_{#1}}%
    #2%
    \IfValueT{#3}{\diff{#3}}%
  }%
}

      }{ % elif {content}
        \IfValueTF{#4}{
          {\left\lVert \hspaceNorm #4 \hspaceNorm \right\rVert}_{#2}%
        }{ % else:
        % \errmessage{norm: Star mode. Don't specify the [size].}%
        \PackageError{FA_DM}{norm: Star mode. Don't specify the [size].}{}%
        }
      }
    }{% elif no [subscript]
      \left\lVert \hspaceNorm #4 \hspaceNorm \right\rVert
    }%
  }{%
    % Non-star version: manual sizing
    \IfValueTF{#2}{% if [size]
      \IfValueTF{#3}{% if [p] 
        {#2\lVert \hspaceNorm #4 \hspaceNorm #2\rVert}_{#3}%
      }{% elif no [p]
        % We assume that #2 was a legal [size] value
        #2\lVert \hspaceNorm #4 \hspaceNorm #2\rVert
      }%
    }{% elif no [size]
      % \errmessage{norm: Size command required. Use \string\norm[\string\big]{F} or \string\norm*{F}}%
      \PackageError{FA_DM}{norm: Size command required. Use \string\norm[\string\big]{F} or \string\norm*{F}}{}%
    }
  }
}
